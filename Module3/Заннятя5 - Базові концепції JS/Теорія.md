## 1. Стрілки (Arrow Functions) — детальніше

```javascript
// Явне повернення (з return)
const add = (a, b) => {
    const result = a + b;
    return result;
};

// Неявне повернення (без return, одне вираження)
const add = (a, b) => a + b;

// Повернення об'єкта — в дужках!
const getUser = () => ({ name: 'Олена', age: 25 });

// this у стрілках — не створює власний this
const obj = {
    name: 'Об\'єкт',
    regular: function() {
        console.log(this.name);  // this = obj
    },
    arrow: () => {
        console.log(this.name);  // this = window (зовнішній контекст)
    }
};
```

## 2. Деструктуризація (Destructuring)

```javascript
// Масиви
const colors = ['red', 'green', 'blue'];
const [first, second, third] = colors;
const [head, ...tail] = colors;  // head='red', tail=['green','blue']
const [a, , c] = colors;         // a='red', c='blue'

// Об'єкти
const user = { name: 'Олена', age: 25, email: 'o@example.com' };
const { name, age } = user;
const { name: userName, email: userEmail } = user;  // з перейменуванням
const { city = 'Невідомо' } = user;  // зі значенням за замовчуванням

// Вкладені об'єкти
const user = {
    name: 'Олена',
    address: { city: 'Київ', street: 'Хрещатик' }
};
const { address: { city, street } } = user;

// У параметрах функції
function printUser({ name, age, email = 'немає' }) {
    console.log(`${name}, ${age}, ${email}`);
}
printUser({ name: 'Іван', age: 30 });
```

## 3. Spread та Rest оператори (...)

```javascript
// Rest — збирає залишок
function logAll(...args) {
    console.log(args);  // [1, 2, 3, 4, 5]
}
logAll(1, 2, 3, 4, 5);

const [first, ...rest] = [1, 2, 3, 4];  // first=1, rest=[2,3,4]
const { name, ...rest } = { name: 'Олена', age: 25, city: 'Київ' };

// Spread — розкладає
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];
const merged = [...arr1, ...arr2];  // [1,2,3,4,5,6]

const obj1 = { a: 1, b: 2 };
const obj2 = { b: 3, c: 4 };
const mergedObj = { ...obj1, ...obj2 };  // {a:1, b:3, c:4}

// Копіювання (не глибоке!)
const copyArr = [...arr1];
const copyObj = { ...obj1 };

// У DOM
const elements = [...document.querySelectorAll('div')];
```

## 4. Promise та Async/Await

### Callback (старий спосіб)
```javascript
setTimeout(() => {
    console.log('Минуло 1 секунда');
}, 1000);
```

### Promise
```javascript
const fetchData = () => {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            const success = true;
            if (success) {
                resolve({ id: 1, name: 'Дані' });
            } else {
                reject(new Error('Помилка завантаження'));
            }
        }, 2000);
    });
};

fetchData()
    .then(data => console.log(data))
    .catch(error => console.error(error))
    .finally(() => console.log('Завершено'));
```

### Async/Await (сучасний спосіб)
```javascript
const loadData = async () => {
    try {
        const data = await fetchData();
        console.log(data);
    } catch (error) {
        console.error('Помилка:', error);
    } finally {
        console.log('Готово');
    }
};
```

### Реальний приклад — fetch API
```javascript
const getUsers = async () => {
    try {
        const response = await fetch('https://jsonplaceholder.typicode.com/users');
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const users = await response.json();
        return users;
    } catch (error) {
        console.error('Не вдалося завантажити користувачів:', error);
        return [];
    }
};

getUsers().then(users => console.table(users));

// POST запит
const createUser = async (userData) => {
    const response = await fetch('https://jsonplaceholder.typicode.com/users', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(userData),
    });
    return response.json();
};
```

## 6. Класи (ES6)

```javascript
class Animal {
    constructor(name) {
        this.name = name;
    }

    speak() {
        console.log(`${this.name} видає звук`);
    }

    static create(name) {
        return new Animal(name);
    }
}

class Dog extends Animal {
    constructor(name, breed) {
        super(name);
        this.breed = breed;
    }

    speak() {
        super.speak();
        console.log('Гав-гав!');
    }
}

const dog = new Dog('Рекс', 'Вівчарка');
dog.speak();

// Приватні поля (новий стандарт)
class User {
    #password;  // приватне поле

    constructor(name, password) {
        this.name = name;
        this.#password = password;
    }

    #encrypt() {  // приватний метод
        return btoa(this.#password);
    }

    getSecureInfo() {
        return { name: this.name, hash: this.#encrypt() };
    }
}
```

## 7. Методи масивів (поглиблено)

```javascript
const items = [
    { id: 1, name: 'Телефон', price: 8000, inStock: true },
    { id: 2, name: 'Ноутбук', price: 35000, inStock: false },
    { id: 3, name: 'Планшет', price: 12000, inStock: true },
];

// map — трансформація
const names = items.map(item => item.name);

// filter — фільтрація
const available = items.filter(item => item.inStock);
const cheap = items.filter(item => item.price < 10000);

// find — перший підходящий
const found = items.find(item => item.id === 2);

// some/every — перевірка
const hasExpensive = items.some(item => item.price > 30000);  // true
const allInStock = items.every(item => item.inStock);          // false

// reduce — агрегація
const total = items.reduce((sum, item) => sum + item.price, 0);
const byStock = items.reduce((acc, item) => {
    acc[item.inStock ? 'inStock' : 'outOfStock'].push(item);
    return acc;
}, { inStock: [], outOfStock: [] });

// sort — сортування
items.sort((a, b) => a.price - b.price);  // за зростанням
items.sort((a, b) => b.price - a.price);  // за спаданням

// chaining — ланцюжок
const result = items
    .filter(item => item.inStock)
    .map(item => ({ ...item, priceWithVat: item.price * 1.2 }))
    .sort((a, b) => a.price - b.price);
```

## 8. Помилки та обробка (try/catch)

```javascript
try {
    const result = riskyOperation();
    console.log(result);
} catch (error) {
    console.error('Сталася помилка:', error.message);
    console.error('Стек:', error.stack);
} finally {
    console.log('Виконується завжди');
}

// Власні помилки
class ValidationError extends Error {
    constructor(message, field) {
        super(message);
        this.name = 'ValidationError';
        this.field = field;
    }
}

function validateUser(data) {
    if (!data.name) throw new ValidationError('Ім\'я обов\'язкове', 'name');
    if (data.age < 0) throw new ValidationError('Вік не може бути від\'ємним', 'age');
}
```
