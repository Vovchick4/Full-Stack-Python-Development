## 1. Що таке React?

React — це JavaScript-бібліотека для створення користувацьких інтерфейсів. Розроблена Facebook у 2013 році.

### Ключові концепції:
- **Компоненти** — перевикористовувані блоки UI
- **JSX** — HTML-подібний синтаксис у JavaScript
- **Virtual DOM** — швидке оновлення тільки змінених частин
- **State** — внутрішній стан компонента
- **Props** — параметри, що передаються в компонент

## 3. Перший компонент

```jsx
// src/components/Greeting.jsx
function Greeting({ name }) {
    return (
        <div className="greeting">
            <h1>Привіт, {name}!</h1>
        </div>
    );
}

export default Greeting;
```

```jsx
// src/App.jsx
import Greeting from './components/Greeting';

function App() {
    return (
        <div>
            <Greeting name="Олена" />
            <Greeting name="Іван" />
        </div>
    );
}

export default App;
```

### JSX — правила

```jsx
// 1. Один кореневий елемент (або Fragment)
function Component() {
    return (
        <>
            <h1>Заголовок</h1>
            <p>Текст</p>
        </>
    );
}

// 2. className замість class
<div className="container">

// 3. JS-вирази в {}
const name = 'Олена';
<h1>Привіт, {name.toUpperCase()}</h1>
<h1>{2 + 2}</h1>
<h1>{condition ? 'Так' : 'Ні'}</h1>

// 4. Стилі — об'єкт
const style = { color: 'red', fontSize: '20px' };
<div style={style}>

// 5. Події — camelCase
<button onClick={() => alert('Клік!')}>Натисни</button>
```

## 4. Компоненти та Props

```jsx
// Компонент з пропсами
function Card({ title, description, price, children }) {
    return (
        <div className="card">
            <h2>{title}</h2>
            <p>{description}</p>
            <span className="price">{price} грн</span>
            {children}
        </div>
    );
}

// Використання
<Card
    title="Футболка"
    description="Зручна бавовняна футболка"
    price={499}
>
    <button>Купити</button>
</Card>

// Props за замовчуванням
function Button({ children, variant = 'primary', disabled = false }) {
    return (
        <button className={`btn btn-${variant}`} disabled={disabled}>
            {children}
        </button>
    );
}
```

## 5. State (Стан)

```jsx
import { useState } from 'react';

function Counter() {
    const [count, setCount] = useState(0);

    return (
        <div>
            <p>Лічильник: {count}</p>
            <button onClick={() => setCount(count + 1)}>+</button>
            <button onClick={() => setCount(count - 1)}>-</button>
            <button onClick={() => setCount(0)}>Скинути</button>
        </div>
    );
}
```

### Правила useState:
1. Не змінюйте state напряму — тільки через setter
2. State оновлюється асинхронно
3. Якщо новий state залежить від старого — використовуйте функцію

```jsx
// Правильно
setCount(prev => prev + 1);

// Для об'єктів — створюйте новий
const [user, setUser] = useState({ name: '', age: 0 });
setUser(prev => ({ ...prev, name: 'Олена' }));
```

## 6. Умовний рендеринг

```jsx
function UserStatus({ user }) {
    if (!user) {
        return <p>Будь ласка, увійдіть</p>;
    }

    return (
        <div>
            <h2>Вітаємо, {user.name}!</h2>
            {user.isAdmin && <p>Панель адміністратора</p>}

                <p>Доступ дозволено</p>
            ) : (
                <p>Доступ заборонено</p>
            )}
        </div>
    );
}
```

## 8. Події в React

```jsx
function Form() {
    const [value, setValue] = useState('');

    const handleSubmit = (e) => {
        e.preventDefault();
        console.log('Відправлено:', value);
    };

    const handleChange = (e) => {
        setValue(e.target.value);
    };

    return (
        <form onSubmit={handleSubmit}>
            <input
                type="text"
                value={value}
                onChange={handleChange}
                placeholder="Введіть текст"
            />
            <button type="submit">Відправити</button>
        </form>
    );
}
```

## 9. Ефекти (useEffect)

```jsx
import { useState, useEffect } from 'react';

function UserList() {
    const [users, setUsers] = useState([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
        const fetchUsers = async () => {
            try {
                const response = await fetch(
                    'https://jsonplaceholder.typicode.com/users'
                );
                if (!response.ok) throw new Error('Помилка завантаження');
                const data = await response.json();
                setUsers(data);
            } catch (err) {
                setError(err.message);
            } finally {
                setLoading(false);
            }
        };

        fetchUsers();
    }, []);  // [] — виконати один раз при монтуванні

    if (loading) return <p>Завантаження...</p>;
    if (error) return <p>Помилка: {error}</p>;

    return (
        <ul>
            {users.map(user => (
                <li key={user.id}>{user.name}</li>
            ))}
        </ul>
    );
}
```
