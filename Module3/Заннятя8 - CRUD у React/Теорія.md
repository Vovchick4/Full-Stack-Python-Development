## 1. Архітектура CRUD у React

CRUD — це Create, Read, Update, Delete. У React ми реалізуємо це через:
- State для зберігання даних
- API-запити для взаємодії з сервером
- Форми для введення даних
- Компоненти для відображення

---

## 2. Підняття State (Lifting State Up)

```jsx
// ✅ Правильно: дані в батьківському компоненті
function App() {
    const [products, setProducts] = useState([]);

    const addProduct = (product) => {
        setProducts(prev => [...prev, { ...product, id: Date.now() }]);
    };

    const deleteProduct = (id) => {
        setProducts(prev => prev.filter(p => p.id !== id));
    };

    const updateProduct = (id, updates) => {
        setProducts(prev => prev.map(p =>
            p.id === id ? { ...p, ...updates } : p
        ));
    };

    return (
        <div>
            <ProductForm onSubmit={addProduct} />
            <ProductList
                products={products}
                onDelete={deleteProduct}
                onUpdate={updateProduct}
            />
        </div>
    );
}
```

## 3. READ — список товарів

```jsx
function ProductList({ products, onDelete, onUpdate }) {
    return (
        <div className="product-grid">
            {products.map(product => (
                <ProductCard
                    key={product.id}
                    product={product}
                    onDelete={onDelete}
                    onUpdate={onUpdate}
                />
            ))}
        </div>
    );
}

function ProductCard({ product, onDelete, onUpdate }) {
    const [isEditing, setIsEditing] = useState(false);

    return (
        <div className="card">
            <img src={product.image} alt={product.title} />
            <h3>{product.title}</h3>
            <p>{product.price} грн</p>
            <div className="actions">
                <button onClick={() => setIsEditing(true)}>
                    Редагувати
                </button>
                <button onClick={() => onDelete(product.id)}>
                    Видалити
                </button>
            </div>
            {isEditing && (
                <EditForm
                    product={product}
                    onSave={(updates) => {
                        onUpdate(product.id, updates);
                        setIsEditing(false);
                    }}
                    onCancel={() => setIsEditing(false)}
                />
            )}
        </div>
    );
}
```

## 4. CREATE — форма додавання

```jsx
function ProductForm({ onSubmit }) {
    const [form, setForm] = useState({
        title: '',
        price: '',
        description: '',
        image: '',
    });
    const [errors, setErrors] = useState({});

    const validate = () => {
        const newErrors = {};
        if (!form.title.trim()) newErrors.title = 'Назва обов\'язкова';
        if (!form.price || form.price <= 0) newErrors.price = 'Ціна має бути > 0';
        if (!form.description.trim()) newErrors.description = 'Опис обов\'язковий';
        return newErrors;
    };

    const handleChange = (e) => {
        const { name, value } = e.target;
        setForm(prev => ({ ...prev, [name]: value }));
        // Очистити помилку при введенні
        if (errors[name]) {
            setErrors(prev => ({ ...prev, [name]: '' }));
        }
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        const newErrors = validate();
        if (Object.keys(newErrors).length > 0) {
            setErrors(newErrors);
            return;
        }
        onSubmit({
            ...form,
            price: Number(form.price),
        });
        setForm({ title: '', price: '', description: '', image: '' });
    };

    return (
        <form className="product-form" onSubmit={handleSubmit}>
            <h2>Додати товар</h2>

            <div className="form-group">
                <label>Назва:</label>
                <input
                    name="title"
                    value={form.title}
                    onChange={handleChange}
                    className={errors.title ? 'error' : ''}
                />
                {errors.title && <span className="error-msg">{errors.title}</span>}
            </div>

            <div className="form-group">
                <label>Ціна (грн):</label>
                <input
                    name="price"
                    type="number"
                    value={form.price}
                    onChange={handleChange}
                />
                {errors.price && <span className="error-msg">{errors.price}</span>}
            </div>

            <div className="form-group">
                <label>Опис:</label>
                <textarea
                    name="description"
                    value={form.description}
                    onChange={handleChange}
                />
                {errors.description && <span className="error-msg">{errors.description}</span>}
            </div>

            <div className="form-group">
                <label>URL зображення:</label>
                <input
                    name="image"
                    value={form.image}
                    onChange={handleChange}
                />
            </div>

            <button type="submit">Додати товар</button>
        </form>
    );
}
```

## 5. UPDATE — редагування

```jsx
function EditForm({ product, onSave, onCancel }) {
    const [form, setForm] = useState({
        title: product.title,
        price: product.price,
        description: product.description,
    });

    const handleSubmit = (e) => {
        e.preventDefault();
        onSave(form);
    };

    return (
        <form className="edit-form" onSubmit={handleSubmit}>
            <h3>Редагувати</h3>
            <input
                value={form.title}
                onChange={e => setForm(prev => ({ ...prev, title: e.target.value }))}
            />
            <input
                type="number"
                value={form.price}
                onChange={e => setForm(prev => ({ ...prev, price: Number(e.target.value) }))}
            />
            <textarea
                value={form.description}
                onChange={e => setForm(prev => ({ ...prev, description: e.target.value }))}
            />
            <div className="edit-actions">
                <button type="submit">Зберегти</button>
                <button type="button" onClick={onCancel}>Скасувати</button>
            </div>
        </form>
    );
}
```

## 6. DELETE — видалення

```jsx
function DeleteButton({ onDelete, itemName }) {
    const [confirm, setConfirm] = useState(false);

    if (confirm) {
        return (
            <div className="confirm-delete">
                <p>Видалити "{itemName}"?</p>
                <button onClick={onDelete}>Так</button>
                <button onClick={() => setConfirm(false)}>Ні</button>
            </div>
        );
    }

    return <button onClick={() => setConfirm(true)}>Видалити</button>;
}
```

### API сервіс
```javascript
// src/api/products.js
const API_URL = 'http://localhost:3001/products';

export const getProducts = async () => {
    const response = await fetch(API_URL);
    if (!response.ok) throw new Error('Помилка завантаження');
    return response.json();
};

export const createProduct = async (product) => {
    const response = await fetch(API_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(product),
    });
    if (!response.ok) throw new Error('Помилка створення');
    return response.json();
};

export const updateProduct = async (id, updates) => {
    const response = await fetch(`${API_URL}/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(updates),
    });
    if (!response.ok) throw new Error('Помилка оновлення');
    return response.json();
};

export const deleteProduct = async (id) => {
    const response = await fetch(`${API_URL}/${id}`, {
        method: 'DELETE',
    });
    if (!response.ok) throw new Error('Помилка видалення');
    return response.json();
};
```

### Компонент з API

```jsx
function App() {
    const [products, setProducts] = useState([]);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    // READ
    useEffect(() => {
        api.getProducts()
            .then(setProducts)
            .catch(setError)
            .finally(() => setLoading(false));
    }, []);

    // CREATE
    const addProduct = async (product) => {
        const newProduct = await api.createProduct(product);
        setProducts(prev => [...prev, newProduct]);
    };

    // UPDATE
    const editProduct = async (id, updates) => {
        const updated = await api.updateProduct(id, updates);
        setProducts(prev => prev.map(p => p.id === id ? updated : p));
    };

    // DELETE
    const removeProduct = async (id) => {
        await api.deleteProduct(id);
        setProducts(prev => prev.filter(p => p.id !== id));
    };

    if (loading) return <Loader />;
    if (error) return <Error message={error.message} />;

    return (
        <div>
            <ProductForm onSubmit={addProduct} />
            <ProductList
                products={products}
                onDelete={removeProduct}
                onUpdate={editProduct}
            />
        </div>
    );
}
```

## 8. Оптимістичне оновлення

```jsx
const removeProduct = async (id) => {
    // Відразу оновлюємо UI
    const previous = products;
    setProducts(prev => prev.filter(p => p.id !== id));

    try {
        await api.deleteProduct(id);
    } catch (error) {
        // Відкочуємо при помилці
        setProducts(previous);
        setError('Не вдалося видалити товар');
    }
};
```
