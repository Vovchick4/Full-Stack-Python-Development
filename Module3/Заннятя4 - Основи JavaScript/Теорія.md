## 1. Що таке JavaScript?

JavaScript (JS) — це мова програмування, яка працює в браузері та дозволяє робити сторінки інтерактивними. Сьогодні JS також працює на сервері (Node.js).

### Підключення JS

```html
<!-- В HTML-файлі (в кінці <body>) -->
<script src="script.js"></script>

<!-- Вбудований скрипт -->
<script>
    console.log('Hello, world!');
</script>

<!-- З async/defer (не блокують завантаження) -->
<script src="script.js" defer></script>
<script src="script.js" async></script>
```

## 2. Змінні та типи даних

```javascript
// var — застарілий, не використовуйте
var oldWay = 'не використовуйте';

// let — змінна (можна змінювати)
let name = 'Олена';
name = 'Марія';  // OK

// const — константа (не можна змінити)
const age = 25;
// age = 30;  // Error!

// Типи даних
const string = 'текст';
const number = 42;
const float = 3.14;
const bool = true;        // або false
const nothing = null;     // нічого (навмисно)
const notDefined = undefined;  // не визначено
const bigInt = 1234567890123456789n;
const symbol = Symbol('id');

// Перевірка типу
typeof 'hello';  // 'string'
typeof 42;       // 'number'
typeof true;     // 'boolean'
typeof null;     // 'object' (історична помилка)
```

## 3. Рядки (Strings)

```javascript
const name = 'Олена';
const greeting = 'Привіт';

// Конкатенація (старий спосіб)
const msg1 = greeting + ', ' + name + '!';

// Шаблонні рядки (новий спосіб)
const msg2 = `${greeting}, ${name}!`;  // "Привіт, Олено!"

// Методи рядків
'hello'.length;              // 5
'hello'.toUpperCase();       // 'HELLO'
'HELLO'.toLowerCase();       // 'hello'
'hello'.indexOf('l');        // 2
'hello'.includes('el');      // true
'  hello  '.trim();          // 'hello'
'hello world'.split(' ');    // ['hello', 'world']
'hello'.slice(1, 4);         // 'ell'
'hello'.replace('l', 'x');   // 'hexlo'
'hello'.replaceAll('l', 'x');// 'hexxo'
```

## 4. Числа та операції

```javascript
const a = 10;
const b = 3;

a + b;   // 13
a - b;   // 7
a * b;   // 30
a / b;   // 3.3333...
a % b;   // 1 (остача)
a ** b;  // 1000 (10³)

// Спеціальні значення
Infinity;     // нескінченність
-Infinity;    
NaN;          // Not a Number (не число)

// Округлення
Math.round(3.7);    // 4
Math.floor(3.7);    // 3
Math.ceil(3.1);     // 4
Math.random();      // випадкове 0-1
Math.max(1, 5, 3);  // 5
```

## 5. Умовні оператори

```javascript
// if/else
const age = 18;

if (age >= 18) {
    console.log('Дорослий');
} else if (age >= 12) {
    console.log('Підліток');
} else {
    console.log('Дитина');
}

// Тернарний оператор
const status = age >= 18 ? 'Дорослий' : 'Дитина';

// switch
const day = 'Monday';
switch (day) {
    case 'Monday':
        console.log('Понеділок');
        break;
    case 'Friday':
        console.log('П\'ятниця!');
        break;
    default:
        console.log('Інший день');
}

// Оператори порівняння
==    // несуворе (5 == '5' -> true!!!)
===   // суворе (5 === '5' -> false)
!=    // несуворе нерівно
!==   // суворе нерівно

// Логічні оператори
&&    // І (обидві true)
||    // АБО (хоча б одна true)
!     // НЕ (заперечення)
??    // nullish (якщо null/undefined, використай запасне)
```

## 6. Масиви (Arrays)

```javascript
const fruits = ['яблуко', 'банан', 'апельсин'];
const mixed = [1, 'text', true, { key: 'value' }];

// Доступ
fruits[0];      // 'яблуко'
fruits.length;  // 3

// Базові методи
fruits.push('груша');       // додати в кінець
fruits.pop();                // видалити з кінця
fruits.unshift('груша');    // додати на початок
fruits.shift();              // видалити з початку
fruits.indexOf('банан');    // 1
fruits.includes('груша');   // false

// Ітерація
for (let i = 0; i < fruits.length; i++) {
    console.log(fruits[i]);
}

for (const fruit of fruits) {
    console.log(fruit);
}

fruits.forEach(fruit => console.log(fruit));

// Трансформація (не змінює оригінал)
const numbers = [1, 2, 3, 4, 5];

numbers.map(n => n * 2);        // [2, 4, 6, 8, 10]
numbers.filter(n => n > 2);     // [3, 4, 5]
numbers.reduce((sum, n) => sum + n, 0);  // 15
numbers.find(n => n > 3);       // 4 (перший)
numbers.every(n => n > 0);      // true (всі > 0)
numbers.some(n => n > 4);       // true (хоч один > 4)
numbers.sort();                  // сортування
```

## 7. Об'єкти (Objects)

```javascript
const user = {
    name: 'Олена',
    age: 25,
    email: 'olena@example.com',
    address: {
        city: 'Київ',
        street: 'Хрещатик',
    },
    greet() {
        console.log(`Привіт, я ${this.name}`);
    },
};

// Доступ
user.name;           // 'Олена'
user['email'];       // 'olena@example.com'
user.address.city;   // 'Київ'

// Зміна
user.age = 26;
user.phone = '+380501234567';

// Видалення
delete user.phone;

// Перевірка
'email' in user;     // true
user.hasOwnProperty('age');  // true

// Ітерація
for (const key in user) {
    console.log(key, user[key]);
}

Object.keys(user);    // ['name', 'age', 'email', 'address']
Object.values(user);  // ['Олена', 25, 'olena@...', {...}]
Object.entries(user); // [['name', 'Олена'], ['age', 25], ...]
```

## 8. Функції

```javascript
// Function declaration (можна викликати до оголошення)
function add(a, b) {
    return a + b;
}

// Function expression (не можна викликати до оголошення)
const multiply = function(a, b) {
    return a * b;
};

// Arrow function (сучасний спосіб)
const subtract = (a, b) => a - b;

// З одним параметром — без дужок
const double = x => x * 2;

// Без параметрів
const greet = () => 'Hello!';

// Параметри за замовчуванням
const greetUser = (name = 'Гість') => `Привіт, ${name}!`;

// Rest параметри
const sum = (...numbers) => numbers.reduce((a, b) => a + b, 0);
sum(1, 2, 3, 4);  // 10
```
