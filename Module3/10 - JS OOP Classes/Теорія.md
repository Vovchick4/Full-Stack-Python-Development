## 1. ООП в JavaScript

Об'єктно-орієнтоване програмування — це парадигма, де дані та логіка grouped в об'єкти.

### 4 принципи ООП:
- **Інкапсуляція** — приховування внутрішніх деталей
- **Наслідування** — створення класів на основі інших
- **Поліморфізм** — однаковий інтерфейс для різних типів
- **Абстракція** — складність прихована за простим інтерфейсом

## 2. Конструктори (старий спосіб)

```javascript
function User(name, age) {
    this.name = name;
    this.age = age;
    this.greet = function() {
        console.log(`Привіт, я ${this.name}`);
    };
}

const user = new User('Олена', 25);
user.greet(); // 'Привіт, я Олена'
```

## 3. Прототипне наслідування (Prototype)

```javascript
function Animal(name) {
    this.name = name;
}

Animal.prototype.speak = function() {
    console.log(`${this.name} видає звук`);
};

function Dog(name, breed) {
    Animal.call(this, name);
    this.breed = breed;
}

// Наслідування
Dog.prototype = Object.create(Animal.prototype);
Dog.prototype.constructor = Dog;

Dog.prototype.speak = function() {
    console.log(`${this.name} гавкає!`);
};

const rex = new Dog('Рекс', 'Вівчарка');
rex.speak(); // 'Рекс гавкає!'
```

## 4. Класи (ES6) — сучасний спосіб

```javascript
class Animal {
    constructor(name) {
        this.name = name;
    }

    speak() {
        console.log(`${this.name} видає звук`);
    }

    // Статичний метод (на класі, не на екземплярі)
    static create(name) {
        return new Animal(name);
    }
}

class Dog extends Animal {
    constructor(name, breed) {
        super(name);
        this.breed = breed;
    }

    speak() {
        super.speak();
        console.log('Гав-гав!');
    }
}
```

## 5. Приватні поля та методи

```javascript
class User {
    #password;  // приватне поле (#)

    constructor(name, password) {
        this.name = name;
        this.#password = password;
    }

    #encrypt() {  // приватний метод
        return btoa(this.#password);
    }

    getSecureInfo() {
        return { name: this.name, hash: this.#encrypt() };
    }

    // Геттер
    get info() {
        return `${this.name}`;
    }

    // Сеттер
    set password(newPass) {
        if (newPass.length < 6) throw new Error('Занадто короткий');
        this.#password = newPass;
    }
}
```

## 6. Шаблони проектування (Design Patterns)

```javascript
// Singleton — єдиний екземпляр
class Database {
    static instance;
    constructor() {
        if (Database.instance) return Database.instance;
        Database.instance = this;
    }
}

// Factory — фабрика об'єктів
class UserFactory {
    static create(type, data) {
        if (type === 'admin') return new Admin(data);
        if (type === 'user') return new RegularUser(data);
    }
}

// Observer — спостерігач
class EventEmitter {
    constructor() {
        this.events = {};
    }
    on(event, listener) {
        if (!this.events[event]) this.events[event] = [];
        this.events[event].push(listener);
    }
    emit(event, data) {
        this.events[event]?.forEach(l => l(data));
    }
}
```

## 7. Функціональне програмування

```javascript
// Чисті функції (не змінюють зовнішній стан)
const add = (a, b) => a + b;

// Immutability (незмінність)
const arr = [1, 2, 3];
const newArr = [...arr, 4];  // створюємо новий, не змінюємо старий

// Higher-order functions (функції вищого порядку)
const multiplyBy = factor => number => number * factor;
const double = multiplyBy(2);
double(5); // 10

// Композиція
const compose = (...fns) => x => fns.reduceRight((acc, fn) => fn(acc), x);
const addTax = price => price * 1.2;
const formatPrice = price => `${price.toFixed(2)} грн`;
const displayPrice = compose(formatPrice, addTax);
displayPrice(100); // '120.00 грн'
```
