## 1. Об'єкти

```javascript
const user = {
    name: 'Олена',
    age: 25,
    email: 'olena@example.com',
    address: { city: 'Київ', street: 'Хрещатик' },
    greet() { console.log(`Привіт, я ${this.name}`); }
};

// Доступ
user.name;          // 'Олена'
user['email'];      // 'olena@...'
user.address.city;  // 'Київ'

// Зміна та додавання
user.age = 26;
user.phone = '+380501234567';

// Видалення
delete user.phone;

// Перевірка
'email' in user;                 // true
user.hasOwnProperty('age');      // true

// Ітерація
Object.keys(user);               // ['name', 'age', 'email', 'address']
Object.values(user);             // значення
Object.entries(user);            // [[key, value], ...]
```

## 3. Контекст (this)

```javascript
// 1. Глобальний контекст (window/globalThis)
console.log(this); // Window (в браузері)

// 2. Метод об'єкта
const obj = {
    name: 'Об\'єкт',
    show() { console.log(this.name); }
};
obj.show(); // 'Об\'єкт'

// 3. Звичайна функція
function show() { console.log(this); }
show(); // Window (або undefined в strict mode)

// 4. Arrow function (не має свого this)
const arrow = () => console.log(this);
// this = зовнішній контекст

// 5. call, apply, bind
function greet(greeting) { console.log(`${greeting}, ${this.name}`); }
const user1 = { name: 'Олена' };
greet.call(user1, 'Привіт');    // 'Привіт, Олена'
greet.apply(user1, ['Привіт']); // 'Привіт, Олена'
const bound = greet.bind(user1); // повертає нову функцію
bound('Привіт');
```

## 4. Замикання (Closure)

Функція "запам'ятовує" змінні з області видимості, де була створена.

```javascript
function createCounter() {
    let count = 0;  // ця змінна "закрита" всередині
    return {
        increment: () => ++count,
        decrement: () => --count,
        getCount: () => count,
    };
}

const counter = createCounter();
counter.increment();  // 1
counter.increment();  // 2
console.log(counter.getCount()); // 2
// count недоступний ззовні!

// Практичний приклад: лічильник
function makeCounter() {
    let count = 0;
    return function() { return ++count; };
}
const c1 = makeCounter();
c1(); // 1
c1(); // 2
```

## 5. Рекурсія

```javascript
// Функція викликає саму себе
function factorial(n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}
factorial(5); // 120

// Обов'язково: базовий випадок (умова виходу)
// Інакше — Stack Overflow!
```

## 6. Spread/Rest оператори (...)

```javascript
// Spread — розкладає
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];
const merged = [...arr1, ...arr2];  // [1,2,3,4,5,6]

const obj1 = { a: 1, b: 2 };
const copy = { ...obj1, c: 3 };     // {a:1, b:2, c:3}

// Копіювання (поверхневе)
const clone = { ...user };
const arrClone = [...arr1];

// Rest — збирає
function logAll(...args) { console.log(args); }
logAll(1, 2, 3); // [1, 2, 3]
```
