## 2. Хуки (Hooks)

### useState — стан
```jsx
const [form, setForm] = useState({ title: '', price: '' });
```

### useEffect — ефекти
```jsx
useEffect(() => {
    // при монтуванні/оновленні
    return () => { /* очищення при демонтажі */ };
}, [dependencies]);
```

### useRef — посилання на DOM
```jsx
const inputRef = useRef(null);
<input ref={inputRef} />
inputRef.current.focus();
```

### useContext — глобальний стан
```jsx
const ThemeContext = createContext('light');
const theme = useContext(ThemeContext);
```

### Кастомні хуки
```jsx
function useFetch(url) {
    const [data, setData] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
        fetch(url)
            .then(res => res.json())
            .then(setData)
            .catch(setError)
            .finally(() => setLoading(false));
    }, [url]);

    return { data, loading, error };
}

// Використання:
const { data: users, loading } = useFetch('/api/users');
```

## 3. Форми в React

```jsx
function ProductForm({ onSubmit }) {
    const [form, setForm] = useState({ title: '', price: '' });
    const [errors, setErrors] = useState({});

    const validate = () => {
        const err = {};
        if (!form.title.trim()) err.title = 'Назва обов\'язкова';
        if (!form.price || form.price <= 0) err.price = 'Ціна > 0';
        return err;
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        const err = validate();
        if (Object.keys(err).length) { setErrors(err); return; }
        onSubmit(form);
        setForm({ title: '', price: '' });
    };

    return (
        <form onSubmit={handleSubmit}>
            <input name="title" value={form.title}
                   onChange={e => setForm(p => ({ ...p, title: e.target.value }))} />
            {errors.title && <span>{errors.title}</span>}
            <input name="price" value={form.price}
                   onChange={e => setForm(p => ({ ...p, price: e.target.value }))} />
            <button type="submit">Створити</button>
        </form>
    );
}
```

## 4. Тестування (Jest + React Testing Library)

```bash
npm install --save-dev @testing-library/react jest
```

```jsx
// __tests__/Button.test.jsx
import { render, screen, fireEvent } from '@testing-library/react';
import Button from './Button';

test('рендерить кнопку з текстом', () => {
    render(<Button>Натисни</Button>);
    expect(screen.getByText('Натисни')).toBeInTheDocument();
});

test('викликає onClick при кліку', () => {
    const onClick = jest.fn();
    render(<Button onClick={onClick}>Клік</Button>);
    fireEvent.click(screen.getByText('Клік'));
    expect(onClick).toHaveBeenCalledTimes(1);
});
```
