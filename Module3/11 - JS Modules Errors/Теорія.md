## 1. Модулі (import/export)

Модулі дозволяють розбивати код на окремі файли.

```javascript
// math.js
export const add = (a, b) => a + b;
export const PI = 3.14159;
export default class Calculator { /* ... */ }

// app.js
import Calculator, { add, PI as pi } from './math.js';
import * as math from './math.js';
math.add(2, 3);

// Експорт за замовчуванням (один на файл)
export default function() { }
```

### Особливості:
- Код в модулях виконується в strict mode
- Кожен модуль має свою область видимості
- import/import() — статичний/динамічний імпорт

## 2. Strict Mode

```javascript
'use strict';  // або в модулях — автоматично

// Зміни в strict mode:
// 1. Заборона неоголошених змінних
x = 10;           // Error (без strict — створить глобальну)

// 2. this = undefined в звичайних функціях
function show() { console.log(this); } // undefined

// 3. Заборона дублювання параметрів
function sum(a, a) { }  // Error

// 4. Заборона delete на змінних
const x = 5;
delete x;  // Error
```

## 3. Обробка помилок (try/catch)

```javascript
try {
    const result = riskyOperation();
    console.log(result);
} catch (error) {
    console.error('Помилка:', error.message);
    console.error('Стек:', error.stack);
} finally {
    console.log('Виконується завжди');
}

// Власні помилки
class ValidationError extends Error {
    constructor(message, field) {
        super(message);
        this.name = 'ValidationError';
        this.field = field;
    }
}

function validateUser(data) {
    if (!data.name) throw new ValidationError('Ім\'я обов\'язкове', 'name');
    if (data.age < 0) throw new ValidationError('Вік не може бути від\'ємним', 'age');
    return true;
}
```

## 4. Cookies

```javascript
// Встановити cookie
document.cookie = 'username=Олена';
document.cookie = 'theme=dark; max-age=3600; path=/';

// Прочитати всі cookies
console.log(document.cookie); // 'username=Олена; theme=dark'

// Функція для отримання cookie
function getCookie(name) {
    const cookies = document.cookie.split('; ');
    for (const cookie of cookies) {
        const [key, value] = cookie.split('=');
        if (key === name) return value;
    }
    return null;
}

// Видалити cookie
document.cookie = 'username=; max-age=0';
```

## 5. ES6/ES7 особливості

```javascript
// ES6 (2015): let, const, arrow functions, classes, template literals,
// destructuring, spread, modules, promises, Map, Set, Symbol

// ES7 (2016):
// Array.includes()
[1, 2, 3].includes(2);  // true
// Exponentiation operator
2 ** 3;  // 8

// ES8 (2017):
// async/await, Object.values(), Object.entries()
// String.padStart(), padEnd()

// ES9 (2018):
// Rest/Spread for objects, Promise.finally()
// RegExp lookbehind assertions

// ES10 (2019):
// Array.flat(), flatMap()
// Object.fromEntries()
// String.trimStart(), trimEnd()

// ES11 (2020):
// Optional chaining (?.), Nullish coalescing (??)
user?.address?.city;     // без помилки якщо user null
const x = value ?? 'default';  // тільки null/undefined
```
