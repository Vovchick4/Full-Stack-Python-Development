## 1. Що таке Django Admin?

Django Admin — це автоматично згенерований інтерфейс для керування даними. Він створюється "з коробки" і дозволяє виконувати CRUD операції з моделями через веб-інтерфейс.

Доступний за адресою: `http://127.0.0.1:8000/admin/`

## 3. Реєстрація моделей

### Проста реєстрація
```python
# blog/admin.py
from django.contrib import admin
from .models import Post, Category, Tag

admin.site.register(Post)
admin.site.register(Category)
admin.site.register(Tag)
```

### Кастомізація через ModelAdmin
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    list_display = ('title', 'author', 'created_at', 'is_published')
    list_filter = ('is_published', 'created_at', 'category')
    search_fields = ('title', 'content')
    ordering = ('-created_at',)
```

## 4. ModelAdmin опції

### Основні налаштування
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    # Колонки для відображення
    list_display = ('title', 'author', 'view_count', 'get_tags', 'created_at')

    # Посилання на деталі (замість ID)
    list_display_links = ('title',)

    # Редагування прямо зі списку
    list_editable = ('is_published',)

    # Фільтри
    list_filter = ('is_published', 'category', 'created_at', 'tags')

    # Пошук
    search_fields = ('title', 'content', 'author__username')

    # Пагінація
    list_per_page = 25

    # Дати
    date_hierarchy = 'created_at'

    # Поля тільки для читання
    readonly_fields = ('view_count', 'created_at', 'updated_at')
```

### Кастомні методи в списку
```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    list_display = ('title', 'short_content', 'created_at')

    def short_content(self, obj):
        return obj.content[:50] + '...' if len(obj.content) > 50 else obj.content
    short_content.short_description = 'Короткий зміст'
```

## 5. Кастомізація форми редагування

```python
@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    # Розділення полів на секції
    fieldsets = (
        ('Основна інформація', {
            'fields': ('title', 'content', 'author')
        }),
        ('Категоризація', {
            'fields': ('category', 'tags'),
            'classes': ('collapse',)  # згорнутий блок
        }),
        ('Публікація', {
            'fields': ('is_published', 'created_at', 'updated_at'),
        }),
    )

    # Виключення полів
    exclude = ('view_count',)

    # Автоматичне заповнення полів
    prepopulated_fields = {'slug': ('title',)}

    # Автозаповнення для ForeignKey
    autocomplete_fields = ('author',)

    # RAW ID для великих таблиць
    raw_id_fields = ('author',)
```

## 6. Inline моделі

Редагування пов'язаних моделей прямо на сторінці батьківської моделі.

```python
class CommentInline(admin.TabularInline):
    model = Comment
    extra = 1  # кількість порожніх форм
    fields = ('author', 'text', 'is_moderated')
    readonly_fields = ('created_at',)

class CommentInline(admin.StackedInline):
    model = Comment
    extra = 1
    # StackedInline — вертикальне відображення
    # TabularInline — табличне відображення

@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    inlines = [CommentInline]
```

## 7. Дії (Actions)

Масові операції з об'єктами.

```python
@admin.action(description="Опублікувати вибрані пости")
def make_published(modeladmin, request, queryset):
    queryset.update(is_published=True)

@admin.action(description="Зняти з публікації")
def make_unpublished(modeladmin, request, queryset):
    queryset.update(is_published=False)

@admin.register(Post)
class PostAdmin(admin.ModelAdmin):
    actions = [make_published, make_unpublished]
```

## 8. Кастомізація адмін-панелі

```python
# Налаштування заголовків
admin.site.site_header = "Мій Блог - Адмінка"
admin.site.site_title = "Адмін-панель"
admin.site.index_title = "Керування контентом"
```

```python
# Кастомна адмін-сторінка
from django.contrib import admin
from django.urls import path
from django.shortcuts import render

class MyAdminSite(admin.AdminSite):
    site_header = "Мій сайт"

    def get_urls(self):
        urls = super().get_urls()
        custom_urls = [
            path('dashboard/', self.admin_view(self.dashboard_view), name='dashboard'),
        ]
        return custom_urls + urls

    def dashboard_view(self, request):
        context = {
            'post_count': Post.objects.count(),
            'user_count': User.objects.count(),
        }
        return render(request, 'admin/dashboard.html', context)
```
