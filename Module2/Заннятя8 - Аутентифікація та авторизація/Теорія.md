## 1. Користувачі та аутентифікація в Django

Django має вбудовану систему аутентифікації з моделлю `User`, яка включає:
- username, email, password
- first_name, last_name
- is_active, is_staff, is_superuser
- date_joined, last_login

## 2. Реєстрація та вхід

### Реєстрація
```python
from django.contrib.auth.forms import UserCreationForm
from django.contrib.auth.models import User
from django.views.generic import CreateView
from django.urls import reverse_lazy

class RegisterView(CreateView):
    model = User
    form_class = UserCreationForm
    template_name = 'registration/register.html'
    success_url = reverse_lazy('login')
```

### Кастомна реєстрація з додатковими полями
```python
from django import forms
from django.contrib.auth.models import User
from django.contrib.auth.forms import UserCreationForm

class CustomUserCreationForm(UserCreationForm):
    email = forms.EmailField(required=True)
    first_name = forms.CharField(max_length=30)
    last_name = forms.CharField(max_length=30)

    class Meta:
        model = User
        fields = ('username', 'email', 'first_name', 'last_name', 'password1', 'password2')

    def save(self, commit=True):
        user = super().save(commit=False)
        user.email = self.cleaned_data['email']
        user.first_name = self.cleaned_data['first_name']
        user.last_name = self.cleaned_data['last_name']
        if commit:
            user.save()
        return user
```

### Вхід/Вихід (вбудовані views)
```python
# mysite/urls.py
from django.urls import path, include

urlpatterns = [
    path('accounts/', include('django.contrib.auth.urls')),
    # Це додає:
    # accounts/login/    [name='login']
    # accounts/logout/   [name='logout']
    # accounts/password_change/  [name='password_change']
    # accounts/password_reset/   [name='password_reset']
]
```

### Кастомний LoginView
```python
from django.contrib.auth.views import LoginView

class CustomLoginView(LoginView):
    template_name = 'registration/login.html'
    redirect_authenticated_user = True  # редирект, якщо вже залогінений
    next_page = 'blog:home'

    def get_form(self, form_class=None):
        form = super().get_form(form_class)
        form.fields['username'].widget.attrs.update({
            'class': 'form-control',
            'placeholder': 'Ім\'я користувача'
        })
        return form
```

## 3. Обмеження доступу

### У функціях (FBV)
```python
from django.contrib.auth.decorators import login_required, permission_required

@login_required(login_url='/login/')
def profile_view(request):
    return render(request, 'profile.html')

@login_required
@permission_required('blog.add_post', raise_exception=True)
def create_post(request):
    # Тільки з правом додавання постів
    pass
```

### У класах (CBV)
```python
from django.contrib.auth.mixins import LoginRequiredMixin, PermissionRequiredMixin, UserPassesTestMixin

class ProfileView(LoginRequiredMixin, TemplateView):
    template_name = 'profile.html'
    login_url = '/login/'

class AdminView(PermissionRequiredMixin, TemplateView):
    permission_required = ('blog.can_publish', 'blog.can_approve')
    template_name = 'admin_panel.html'

class AuthorDashboardView(UserPassesTestMixin, TemplateView):
    template_name = 'dashboard.html'

    def test_func(self):
        return self.request.user.groups.filter(name='Authors').exists()
```

## 4. Розширення моделі User

### Спосіб 1: OneToOneField (Profile)
```python
from django.db import models
from django.contrib.auth.models import User

class Profile(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE, related_name='profile')
    bio = models.TextField(blank=True)
    birth_date = models.DateField(null=True, blank=True)
    avatar = models.ImageField(upload_to='avatars/', blank=True)

    def __str__(self):
        return f"Profile: {self.user.username}"
```

### Сигнал для автоматичного створення профілю
```python
# signals.py
from django.db.models.signals import post_save
from django.dispatch import receiver
from django.contrib.auth.models import User

@receiver(post_save, sender=User)
def create_user_profile(sender, instance, created, **kwargs):
    if created:
        Profile.objects.create(user=instance)

@receiver(post_save, sender=User)
def save_user_profile(sender, instance, **kwargs):
    instance.profile.save()
```

### Спосіб 2: AbstractUser (кастомна модель User)
```python
# models.py
from django.contrib.auth.models import AbstractUser
from django.db import models

class CustomUser(AbstractUser):
    bio = models.TextField(blank=True)
    birth_date = models.DateField(null=True, blank=True)
    phone = models.CharField(max_length=15, blank=True)

    def __str__(self):
        return self.email

# settings.py
AUTH_USER_MODEL = 'myapp.CustomUser'
```

## 5. Групи та дозволи

```python
from django.contrib.auth.models import Group, Permission
from django.contrib.contenttypes.models import ContentType

# Створення груп
editor_group, created = Group.objects.get_or_create(name='Editors')

# Додавання дозволів
content_type = ContentType.objects.get_for_model(Post)
permission = Permission.objects.get(
    codename='can_publish',
    content_type=content_type,
)
editor_group.permissions.add(permission)

# Додавання користувача в групу
user.groups.add(editor_group)
```

### Кастомні дозволи в моделі
```python
class Post(models.Model):
    class Meta:
        permissions = [
            ('can_publish', 'Can publish posts'),
            ('can_approve', 'Can approve comments'),
        ]
# Потім: python manage.py makemigrations && python manage.py migrate
```

## 6. Шаблони для аутентифікації

### login.html
```html
{% if form.errors %}
    <p>Невірне ім'я користувача або пароль.</p>
{% endif %}

<form method="post">
    {% csrf_token %}
    {{ form.as_p }}
    <button type="submit">Увійти</button>
</form>

<a href="{% url 'password_reset' %}">Забули пароль?</a>
```

### Перевірка в шаблоні
```html
{% if user.is_authenticated %}
    <p>Вітаємо, {{ user.username }}!</p>
    <a href="{% url 'logout' %}">Вийти</a>
{% else %}
    <a href="{% url 'login' %}">Увійти</a>
    <a href="{% url 'register' %}">Реєстрація</a>
{% endif %}

{% if perms.blog.add_post %}
    <a href="{% url 'blog:create' %}">Створити пост</a>
{% endif %}
```
